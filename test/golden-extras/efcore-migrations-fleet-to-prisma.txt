generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

model Vehicle {
  id            Int      @id @default(autoincrement()) @map("Id")
  plate         String   @map("Plate") @db.Text
  discriminator String   @map("Discriminator") @db.VarChar(8)
  seats         Int?     @map("Seats")
  payload       Decimal? @map("Payload")
  axles         Int?     @map("Axles")

  @@map("Vehicles")
}
