import { z } from 'zod';

export const Role = {
  ADMIN: 'admin',
  USER: 'user',
} as const;
export const RoleSchema = z.enum(Role);
export type Role = z.infer<typeof RoleSchema>;

export const OrderSchema = z.object({
  region: z.string(),
  number: z.number().int(),
  total: z.string().regex(/^-?\d+(\.\d+)?$/),
});
export type Order = z.infer<typeof OrderSchema>;

export const OrderCreateSchema = OrderSchema;
export type OrderCreate = z.infer<typeof OrderCreateSchema>;

export const OrderUpdateSchema = OrderCreateSchema.partial();
export type OrderUpdate = z.infer<typeof OrderUpdateSchema>;

export const OrderLineSchema = z.object({
  id: z.number().int(),
  orderRegion: z.string(),
  orderNumber: z.number().int(),
  sku: z.string().max(32),
});
export type OrderLine = z.infer<typeof OrderLineSchema>;

export const OrderLineCreateSchema = OrderLineSchema.omit({ id: true });
export type OrderLineCreate = z.infer<typeof OrderLineCreateSchema>;

export const OrderLineUpdateSchema = OrderLineCreateSchema.partial();
export type OrderLineUpdate = z.infer<typeof OrderLineUpdateSchema>;

export const AccountSchema = z.object({
  id: z.number().int(),
  email: z.string(),
  tenantId: z.number().int(),
});
export type Account = z.infer<typeof AccountSchema>;

export const AccountCreateSchema = AccountSchema.omit({ id: true });
export type AccountCreate = z.infer<typeof AccountCreateSchema>;

export const AccountUpdateSchema = AccountCreateSchema.partial();
export type AccountUpdate = z.infer<typeof AccountUpdateSchema>;

export type AccountWithRelations = Account & {
  tenant?: TenantWithRelations;
  profile?: ProfileWithRelations | null;
};
export const AccountWithRelationsSchema: z.ZodType<AccountWithRelations> = AccountSchema.extend({
  tenant: z.lazy(() => TenantWithRelationsSchema).optional(),
  profile: z.lazy(() => ProfileWithRelationsSchema).nullable().optional(),
});

export const TenantSchema = z.object({
  id: z.number().int(),
});
export type Tenant = z.infer<typeof TenantSchema>;

export const TenantCreateSchema = TenantSchema.omit({ id: true });
export type TenantCreate = z.infer<typeof TenantCreateSchema>;

export const TenantUpdateSchema = TenantCreateSchema.partial();
export type TenantUpdate = z.infer<typeof TenantUpdateSchema>;

export type TenantWithRelations = Tenant & {
  accounts?: AccountWithRelations[];
};
export const TenantWithRelationsSchema: z.ZodType<TenantWithRelations> = TenantSchema.extend({
  accounts: z.array(z.lazy(() => AccountWithRelationsSchema)).optional(),
});

export const ProfileSchema = z.object({
  region: z.string(),
  number: z.number().int(),
  accountId: z.number().int(),
});
export type Profile = z.infer<typeof ProfileSchema>;

export const ProfileCreateSchema = ProfileSchema;
export type ProfileCreate = z.infer<typeof ProfileCreateSchema>;

export const ProfileUpdateSchema = ProfileCreateSchema.partial();
export type ProfileUpdate = z.infer<typeof ProfileUpdateSchema>;

export type ProfileWithRelations = Profile & {
  account?: AccountWithRelations;
};
export const ProfileWithRelationsSchema: z.ZodType<ProfileWithRelations> = ProfileSchema.extend({
  account: z.lazy(() => AccountWithRelationsSchema).optional(),
});

export const DocumentSchema = z.object({
  id: z.string(),
  ulid: z.string(),
  short: z.string(),
  uid: z.uuid(),
  token: z.uuid(),
  title: z.string(),
  body: z.string(),
  tags: z.array(z.string()),
  scores: z.array(z.number().int()),
  role: RoleSchema,
  small: z.number().int(),
  ratio: z.number(),
  price: z.string().regex(/^-?\d{1,8}(\.\d{1,2})?$/),
  cost: z.string().regex(/^-?\d+(\.\d+)?$/),
  views: z.string().regex(/^-?\d+$/),
  blob: z.base64(),
  meta: z.unknown(),
  created: z.coerce.date(),
  updated: z.coerce.date(),
  day: z.coerce.date(),
  at: z.string(),
  ip: z.union([z.ipv4(), z.ipv6()]),
  area: z.unknown().nullable(),
  place: z.unknown(),
  window: z.object({ lower: z.number().int().nullable(), upper: z.number().int().nullable(), bounds: z.string().optional() }).nullable(),
  internal: z.number().int(),
  legacy: z.string(),
});
export type Document = z.infer<typeof DocumentSchema>;

export const DocumentCreateSchema = DocumentSchema.omit({ updated: true }).partial({ id: true, ulid: true, short: true, uid: true, token: true, scores: true, role: true, created: true, area: true, place: true, window: true });
export type DocumentCreate = z.infer<typeof DocumentCreateSchema>;

export const DocumentUpdateSchema = DocumentCreateSchema.partial();
export type DocumentUpdate = z.infer<typeof DocumentUpdateSchema>;

export const ScratchSchema = z.object({
  id: z.number().int(),
});
export type Scratch = z.infer<typeof ScratchSchema>;

export const ScratchCreateSchema = ScratchSchema;
export type ScratchCreate = z.infer<typeof ScratchCreateSchema>;

export const ScratchUpdateSchema = ScratchCreateSchema.partial();
export type ScratchUpdate = z.infer<typeof ScratchUpdateSchema>;

export const DocumentSummarySchema = z.object({
  id: z.string(),
  title: z.string(),
});
export type DocumentSummary = z.infer<typeof DocumentSummarySchema>;
