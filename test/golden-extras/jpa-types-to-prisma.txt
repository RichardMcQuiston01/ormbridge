generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

enum Level {
  LOW
  MEDIUM
  HIGH
}

enum Color {
  RED @map("0")
  GREEN @map("1")
  BLUE @map("2")
}

model Sample {
  id                   BigInt            @id @default(autoincrement())
  plain                String?           @db.VarChar(255)
  required             String            @db.VarChar(255)
  sized                String?           @db.VarChar(40)
  validated            String            @db.VarChar(30)
  blank                String?           @db.VarChar(255)
  nonEmpty             String?           @db.VarChar(255)
  email                String?           @db.VarChar(255)
  code                 String?           @unique @db.VarChar(255)
  basicRequired        String            @db.VarChar(255)
  naturalKey           String?           @unique @db.VarChar(255)
  commented            String?           @db.VarChar(255)
  count                Int
  maybeCount           Int?
  small                Int
  tiny                 Int?
  big                  BigInt
  flag                 Boolean
  maybeFlag            Boolean?
  single               Float
  twice                Float
  money                Decimal?          @db.Decimal(38, 2)
  precise              Decimal?          @db.Decimal(10, 3)
  digits               Decimal?          @db.Decimal(6, 1)
  huge                 Decimal?          @db.Decimal(38, 0)
  letter               String            @db.VarChar(1)
  maybeLetter          String?           @db.VarChar(1)
  answer               Int               @default(42)
  day                  DateTime?         @db.Date
  time                 DateTime?         @db.Time(6)
  stamp                DateTime?         @db.Timestamptz(6)
  instant              DateTime?         @db.Timestamptz(6)
  offset               DateTime?         @db.Timestamptz(6)
  zoned                DateTime?         @db.Timestamptz(6)
  duration             BigInt?
  legacyDate           DateTime?         @db.Date
  legacyStamp          DateTime?         @db.Timestamptz(6)
  legacyTime           DateTime?         @db.Time(6)
  calendar             DateTime?         @db.Timestamptz(6)
  sqlDate              DateTime?         @db.Date
  sqlStamp             DateTime?         @db.Timestamptz(6)
  uuid                 String?           @db.Uuid
  bytes                Bytes?
  blob                 Bytes?
  clob                 String?           @db.Text
  handle               Bytes?
  json                 Json?
  node                 Json?
  level                Level?
  color                Color?
  price                String?           @db.VarChar(255)
  legacyDefinition     String            @default("x") @db.VarChar(12)
  version              BigInt
  sample_nicknames_set SampleNicknames[]
  sample_levels_set    SampleLevels[]
  sample_ordered_set   SampleOrdered[]
}

model Widget {
  id BigInt @id

  @@map("Gadget")
}

model SampleNicknames {
  nicknames String? @db.VarChar(255)
  Sample_id BigInt
  sample    Sample  @relation(fields: [Sample_id], references: [id], onDelete: NoAction)

  @@map("Sample_nicknames")
}

model SampleLevels {
  levels    Level?
  Sample_id BigInt
  sample    Sample @relation(fields: [Sample_id], references: [id], onDelete: NoAction)

  @@map("Sample_levels")
}

model SampleOrdered {
  ordered      String? @db.VarChar(255)
  orderedOrder Int     @map("ordered_ORDER")
  Sample_id    BigInt
  sample       Sample  @relation(fields: [Sample_id], references: [id], onDelete: NoAction)

  @@id([Sample_id, orderedOrder])
  @@map("Sample_ordered")
}
