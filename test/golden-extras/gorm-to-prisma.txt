generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

enum Role {
  Customer @map("customer")
  Staff @map("staff")
  Admin @map("admin")
}

model Customer {
  id             BigInt    @id @default(autoincrement())
  createdAt      DateTime  @default(now()) @map("created_at") @db.Timestamptz(6)
  updatedAt      DateTime  @updatedAt @map("updated_at") @db.Timestamptz(6)
  deletedAt      DateTime? @map("deleted_at") @db.Timestamptz(6)
  email          String    @unique @db.VarChar(255)
  name           String?   @db.VarChar(100)
  role           Role      @default(Customer)
  billingStreet  String    @map("billing_street") @db.VarChar(120)
  billingCity    String    @map("billing_city") @db.VarChar(60)
  billingPostal  String?   @map("billing_zip") @db.VarChar(12)
  shippingStreet String    @map("shipping_street") @db.VarChar(120)
  shippingCity   String    @map("shipping_city") @db.VarChar(60)
  shippingPostal String?   @map("shipping_zip") @db.VarChar(12)
  balance        Decimal   @default(0) @db.Decimal(12, 2)
  rank           BigInt
  avatar         Bytes
  settings       Json
  orders         Order[]

  @@index([deletedAt])
  @@map("customers")
}

model Order {
  id          String        @id @default(uuid()) @db.Uuid
  placedAt    DateTime      @default(now()) @map("placed_at") @db.Timestamptz(6)
  modifiedAt  BigInt        @map("modified_at")
  total       Decimal       @db.Decimal(12, 2)
  note        String        @db.VarChar(500)
  customer_id BigInt
  customer    Customer      @relation(fields: [customer_id], references: [id], onDelete: Restrict, onUpdate: Cascade)
  items       OrderItem[]
  labels      OrderLabels[]

  @@index([customer_id, placedAt(sort: Desc)], map: "idx_order_customer_placed")
  @@map("shop_orders")
}

model OrderItem {
  lineNo      Int     @map("line_no")
  quantity    Int     @default(1)
  product_sku String  @db.VarChar(40)
  product     Product @relation(fields: [product_sku], references: [sku], onDelete: NoAction)
  order_id    String  @db.Uuid
  order       Order   @relation(fields: [order_id], references: [id], onDelete: NoAction)

  @@id([order_id, lineNo])
  @@map("order_items")
}

model Product {
  sku             String           @id @db.VarChar(40)
  title           String           @db.VarChar(200)
  slug            String           @db.VarChar(100)
  price           Decimal          @db.Decimal(10, 2)
  stock           Int              @default(0) @map("in_stock")
  deletedAt       DateTime?        @map("deleted_at") @db.Timestamptz(6)
  order_item_set  OrderItem[]
  similar         ProductSimilar[] @relation("ProductSimilar_from_product")
  product_similar ProductSimilar[] @relation("ProductSimilar_to_product")

  @@index([title])
  @@index([slug], type: Hash, map: "idx_product_slug")
  @@map("products")
}

model Label {
  id           BigInt        @id @default(autoincrement())
  name         String        @unique @db.VarChar(40)
  order_labels OrderLabels[]

  @@map("labels")
}

model Comment {
  id        BigInt @id @default(autoincrement())
  body      String @db.Text
  ownerId   BigInt @map("owner_id")
  ownerType String @map("owner_type") @db.VarChar(30)

  @@map("comments")
}

model OrderLabels {
  id       Int    @id @default(autoincrement())
  order_id String @db.Uuid
  order    Order  @relation(fields: [order_id], references: [id], onDelete: Cascade)
  label_id BigInt
  label    Label  @relation(fields: [label_id], references: [id], onDelete: Cascade)

  @@unique([order_id, label_id])
  @@map("shop_orders_labels")
}

model ProductSimilar {
  id              Int     @id @default(autoincrement())
  from_product_id String  @db.VarChar(40)
  from_product    Product @relation("ProductSimilar_from_product", fields: [from_product_id], references: [sku], onDelete: Cascade)
  to_product_id   String  @db.VarChar(40)
  to_product      Product @relation("ProductSimilar_to_product", fields: [to_product_id], references: [sku], onDelete: Cascade)

  @@unique([from_product_id, to_product_id])
  @@map("products_similar")
}
