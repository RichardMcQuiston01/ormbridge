generator client {
  provider        = "prisma-client-js"
  previewFeatures = ["views"]
}

datasource db {
  provider = "sqlserver"
  url      = env("DATABASE_URL")
  schemas  = ["public", "audit"]
}

model Order {
  region String
  number Int
  total  Decimal
  lines  OrderLine[]

  @@id([region, number], map: "order_pkey")
  @@schema("public")
}

model OrderLine {
  id          Int    @id @default(autoincrement())
  orderRegion String
  orderNumber Int
  sku         String @db.VarChar(32)
  order       Order  @relation(fields: [orderRegion, orderNumber], references: [region, number], onDelete: Cascade, onUpdate: NoAction, map: "fk_line_order")

  @@index([orderRegion, orderNumber])
  @@schema("public")
}

model Account {
  id       Int      @id(map: "account_pk") @default(autoincrement())
  email    String   @unique(map: "uq_account_email")
  tenantId Int
  tenant   Tenant   @relation(fields: [tenantId], references: [id], onDelete: NoAction, onUpdate: Cascade, map: "fk_account_tenant")
  profile  Profile?

  @@schema("public")
}

model Tenant {
  id       Int       @id @default(autoincrement())
  accounts Account[]

  @@schema("public")
}

model Profile {
  region    String
  number    Int
  accountId Int     @unique
  account   Account @relation(fields: [accountId], references: [id], onDelete: NoAction)

  @@id([region, number])
  @@schema("public")
}

model Document {
  id       String                               @id @default(cuid())
  ulid     String                               @default(ulid()) @unique(map: "uq_document_ulid")
  short    String                               @default(nanoid(12))
  uid      String                               @default(uuid(7)) @db.UniqueIdentifier
  token    String                               @default(dbgenerated("gen_random_uuid()")) @db.UniqueIdentifier
  title    String
  body     String                               @db.Text
  tags     String                               @db.NVarChar(Max)
  scores   String                               @default("[]") @db.NVarChar(Max)
  role     String                               @default("user")
  small    Int                                  @db.SmallInt
  ratio    Float                                @db.Real
  price    Decimal                              @db.Decimal(10, 2)
  cost     Decimal
  views    BigInt
  blob     Bytes
  meta     String                               @db.NVarChar(Max)
  created  DateTime                             @default(now())
  updated  DateTime                             @updatedAt
  day      DateTime                             @db.Date
  at       DateTime                             @db.Time
  ip       String
  area     Unsupported("circle")?
  place    Unsupported("geometry(Point, 4326)") @default(dbgenerated("'POINT(0 0)'::geometry"))
  window   String?
  internal Int                                  @ignore
  legacy   String                               @map("legacy_name")

  @@index([title(sort: Desc), created], map: "ix_document_title")
  @@index([meta])
  @@unique([title, day], map: "uq_document_title_day")
  @@map("documents")
  @@schema("audit")
}

model Scratch {
  id Int @id

  @@ignore
  @@schema("audit")
}

view DocumentSummary {
  id    String @unique
  title String

  @@schema("audit")
}
