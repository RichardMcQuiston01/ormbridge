generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

model Vehicle {
  id           Int      @id @default(autoincrement()) @map("Id")
  plate        String   @unique @map("Plate") @db.Text
  registeredOn DateTime @map("RegisteredOn") @db.Timestamptz(6)
  doors        Int?     @map("Doors")
  capacity     Decimal? @map("Capacity")
  trailer      String?  @map("Trailer") @db.Text
  kind         String   @map("Kind") @db.Text

  @@map("Vehicles")
}
