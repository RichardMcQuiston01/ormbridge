generator client {
  provider        = "prisma-client-js"
  previewFeatures = ["views"]
}

datasource db {
  provider = "mysql"
  url      = env("DATABASE_URL")
}

enum Role {
  ADMIN @map("admin")
  USER @map("user")

  @@map("role_t")
}

model Order {
  region String
  number Int
  total  Decimal
  lines  OrderLine[]

  @@id([region, number])
}

model OrderLine {
  id          Int    @id @default(autoincrement())
  orderRegion String
  orderNumber Int
  sku         String @db.VarChar(32)
  order       Order  @relation(fields: [orderRegion, orderNumber], references: [region, number], onDelete: Cascade, onUpdate: Restrict, map: "fk_line_order")

  @@index([orderRegion, orderNumber])
}

model Account {
  id       Int      @id @default(autoincrement())
  email    String   @unique(map: "uq_account_email")
  tenantId Int
  tenant   Tenant   @relation(fields: [tenantId], references: [id], onDelete: NoAction, onUpdate: Cascade, map: "fk_account_tenant")
  profile  Profile?
}

model Tenant {
  id       Int       @id @default(autoincrement())
  accounts Account[]
}

model Profile {
  region    String
  number    Int
  accountId Int     @unique
  account   Account @relation(fields: [accountId], references: [id], onDelete: Restrict)

  @@id([region, number])
}

model Document {
  id       String                               @id @default(cuid())
  ulid     String                               @default(ulid()) @unique(map: "uq_document_ulid")
  short    String                               @default(nanoid(12))
  uid      String                               @default(uuid(7))
  token    String                               @default(dbgenerated("gen_random_uuid()"))
  title    String
  body     String                               @db.Text
  tags     Json
  scores   Json                                 @default("[]")
  role     Role                                 @default(USER)
  small    Int                                  @db.SmallInt
  ratio    Float
  price    Decimal                              @db.Decimal(10, 2)
  cost     Decimal
  views    BigInt
  blob     Bytes
  meta     Json
  created  DateTime                             @default(now())
  updated  DateTime                             @updatedAt @db.Timestamp(0)
  day      DateTime                             @db.Date
  at       DateTime                             @db.Time(6)
  ip       String
  area     Unsupported("circle")?
  place    Unsupported("geometry(Point, 4326)") @default(dbgenerated("'POINT(0 0)'::geometry"))
  window   String?
  internal Int                                  @ignore
  legacy   String                               @map("legacy_name")

  @@index([title(sort: Desc), created], map: "ix_document_title")
  @@index([meta])
  @@unique([title, day], map: "uq_document_title_day")
  @@map("documents")
}

model Scratch {
  id Int @id

  @@ignore
}

view DocumentSummary {
  id    String @unique
  title String
}
