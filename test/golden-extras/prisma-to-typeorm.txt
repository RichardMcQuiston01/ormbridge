import { Column, CreateDateColumn, Entity, Index, JoinColumn, ManyToOne, OneToMany, OneToOne, PrimaryColumn, PrimaryGeneratedColumn, type Relation, Unique, UpdateDateColumn, ViewColumn, ViewEntity } from 'typeorm';

export enum Role {
  ADMIN = 'admin',
  USER = 'user',
}

@Entity('Order')
export class Order {
  @PrimaryColumn({ type: 'varchar' })
  region!: string;

  @PrimaryColumn({ type: 'int' })
  number!: number;

  @Column({ type: 'decimal' })
  total!: string;
}

@Entity('OrderLine')
@Index(['orderRegion', 'orderNumber'])
export class OrderLine {
  @PrimaryGeneratedColumn('increment')
  id!: number;

  @Column({ type: 'varchar' })
  orderRegion!: string;

  @Column({ type: 'int' })
  orderNumber!: number;

  @Column({ type: 'varchar', length: 32 })
  sku!: string;
}

@Entity('Account')
export class Account {
  @PrimaryGeneratedColumn('increment')
  id!: number;

  @Column({ type: 'varchar', unique: true })
  email!: string;

  @ManyToOne(() => Tenant, (entity) => entity.accounts, {
    onDelete: 'NO ACTION',
    nullable: false,
  })
  @JoinColumn({ name: 'tenantId' })
  tenant!: Relation<Tenant>;

  @OneToOne(() => Profile, (entity) => entity.account)
  profile!: Relation<Profile> | null;
}

@Entity('Tenant')
export class Tenant {
  @PrimaryGeneratedColumn('increment')
  id!: number;

  @OneToMany(() => Account, (entity) => entity.tenant)
  accounts!: Account[];
}

@Entity('Profile')
export class Profile {
  @PrimaryColumn({ type: 'varchar' })
  region!: string;

  @PrimaryColumn({ type: 'int' })
  number!: number;

  @OneToOne(() => Account, (entity) => entity.profile, {
    onDelete: 'RESTRICT',
    nullable: false,
  })
  @JoinColumn({ name: 'accountId' })
  account!: Relation<Account>;
}

@Entity('documents')
@Index('ix_document_title', ['title', 'created'])
@Index(['meta'])
@Unique('uq_document_title_day', ['title', 'day'])
export class Document {
  @PrimaryColumn({ type: 'varchar' })
  id!: string;

  @Column({ type: 'varchar', unique: true })
  ulid!: string;

  @Column({ type: 'varchar' })
  short!: string;

  @Column({ type: 'uuid', default: () => 'gen_random_uuid()' })
  uid!: string;

  @Column({ type: 'uuid' })
  token!: string;

  @Column({ type: 'varchar' })
  title!: string;

  @Column({ type: 'text' })
  body!: string;

  @Column({ type: 'varchar', array: true })
  tags!: string[];

  @Column({ type: 'int', array: true, default: () => "'{}'" })
  scores!: number[];

  @Column({ type: 'enum', enum: Role, default: Role.USER })
  role!: Role;

  @Column({ type: 'int' })
  small!: number;

  @Column({ type: 'float' })
  ratio!: number;

  @Column({ type: 'decimal', precision: 10, scale: 2 })
  price!: string;

  @Column({ type: 'decimal' })
  cost!: string;

  @Column({ type: 'bigint' })
  views!: string;

  @Column({ type: 'bytea' })
  blob!: Buffer;

  @Column({ type: 'jsonb' })
  meta!: unknown;

  @CreateDateColumn({ type: 'timestamptz' })
  created!: Date;

  @UpdateDateColumn({ type: 'timestamptz' })
  updated!: Date;

  @Column({ type: 'date' })
  day!: string;

  @Column({ type: 'time' })
  at!: string;

  @Column({ type: 'inet' })
  ip!: string;

  @Column({ type: 'varchar', nullable: true })
  area!: string | null;

  @Column({ type: 'varchar' })
  place!: string;

  @Column({ type: 'int4range', nullable: true })
  window!: string | null;

  @Column({ type: 'int' })
  internal!: number;

  @Column({ name: 'legacy_name', type: 'varchar' })
  legacy!: string;
}

@Entity('Scratch')
export class Scratch {
  @PrimaryColumn({ type: 'int' })
  id!: number;
}
