generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

enum PostStatus {
  DRAFT @map("draft")
  PUBLISHED @map("published")

  @@map("post_status")
}

model BlogUser {
  id              String       @id @default(uuid()) @db.Uuid
  createdAt       DateTime     @default(now()) @map("created_at") @db.Timestamptz(6)
  updatedAt       DateTime     @updatedAt @map("updated_at") @db.Timestamptz(6)
  authorBlogPosts BlogPost[]   @relation("BlogPost_author")
  editorBlogPosts BlogPost[]   @relation("BlogPost_editor")
  blogProfile     BlogProfile?

  @@map("blog_user")
}

model BlogCategory {
  id             String         @id @default(uuid()) @db.Uuid
  createdAt      DateTime       @default(now()) @map("created_at") @db.Timestamptz(6)
  updatedAt      DateTime       @map("updated_at") @db.Timestamptz(6)
  name           String         @unique @db.VarChar(100)
  slug           String         @db.VarChar(50)
  parentId       String?        @map("parent_id") @db.Uuid
  parent         BlogCategory?  @relation("BlogCategory_parent", fields: [parentId], references: [id], onDelete: SetNull)
  blogCategories BlogCategory[] @relation("BlogCategory_parent")
  blogPosts      BlogPost[]

  @@map("blog_category")
}

model BlogTag {
  id        String     @id @default(uuid()) @db.Uuid
  label     String     @db.VarChar(50)
  createdAt DateTime   @default(now()) @map("created_at") @db.Timestamptz(6)
  updatedAt DateTime   @updatedAt @map("updated_at") @db.Timestamptz(6)
  blogPosts BlogPost[]

  @@map("blog_tag")
}

model BlogPost {
  id          String       @id @default(uuid()) @db.Uuid
  createdAt   DateTime     @default(now()) @map("created_at") @db.Timestamptz(6)
  updatedAt   DateTime     @map("updated_at") @db.Timestamptz(6)
  publicId    String       @default(uuid()) @unique @map("public_id") @db.Uuid
  title       String       @db.VarChar(200)
  body        String       @db.Text
  status      PostStatus   @default(DRAFT)
  rating      Decimal?     @db.Decimal(4, 2)
  viewCount   Int          @default(0) @map("view_count")
  isFeatured  Boolean      @default(false) @map("is_featured")
  publishedAt DateTime     @default(now()) @map("published_at") @db.Timestamptz(6)
  metadata    Json         @default("{}")
  authorId    String       @map("author_id") @db.Uuid
  author      BlogUser     @relation("BlogPost_author", fields: [authorId], references: [id], onDelete: Cascade)
  editorId    String?      @map("editor_id") @db.Uuid
  editor      BlogUser?    @relation("BlogPost_editor", fields: [editorId], references: [id], onDelete: SetNull)
  categoryId  String       @map("category_id") @db.Uuid
  category    BlogCategory @relation(fields: [categoryId], references: [id], onDelete: Restrict, map: "blog_post_category_id_fkey")
  tags        BlogTag[]

  @@unique([authorId, title], map: "blog_post_author_id_title_key")
  @@index([title], map: "blog_post_title_idx")
  @@index([publishedAt, status], map: "post_pub_status_idx")
  @@map("blog_post")
}

model BlogProfile {
  id        String   @id @default(uuid()) @db.Uuid
  bio       String?  @db.Text
  avatar    String?  @db.VarChar(100)
  createdAt DateTime @default(now()) @map("created_at") @db.Timestamptz(6)
  updatedAt DateTime @updatedAt @map("updated_at") @db.Timestamptz(6)
  userId    String   @unique @map("user_id") @db.Uuid
  user      BlogUser @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@map("blog_profile")
}
