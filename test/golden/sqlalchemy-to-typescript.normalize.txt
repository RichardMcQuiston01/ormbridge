export enum PostStatus {
  DRAFT = 'draft',
  PUBLISHED = 'published',
}

export interface User {
  id: string;
  createdAt: string;
  updatedAt: string;
  posts?: Post[];
  editedPosts?: Post[];
  profile?: Profile | null;
}

export interface Category {
  createdAt: string;
  updatedAt: string;
  id: string;
  name: string;
  slug: string;
  parentId: string | null;
  parent?: Category | null;
  children?: Category[];
  posts?: Post[];
}

export interface Post {
  createdAt: string;
  updatedAt: string;
  id: string;
  publicId: string;
  title: string;
  body: string;
  status: PostStatus;
  rating: string | null;
  viewCount: number;
  isFeatured: boolean;
  publishedAt: string;
  metadata: unknown;
  authorId: string;
  author?: User;
  editorId: string | null;
  editor?: User | null;
  categoryId: string;
  category?: Category;
  tags?: Tag[];
}

export interface Tag {
  id: string;
  label: string;
  createdAt: string;
  updatedAt: string;
  posts?: Post[];
}

export interface Profile {
  id: string;
  bio: string | null;
  avatar: string | null;
  createdAt: string;
  updatedAt: string;
  userId: string;
  user?: User;
}
