import { Column, CreateDateColumn, Entity, Index, JoinColumn, JoinTable, ManyToMany, ManyToOne, OneToMany, OneToOne, PrimaryGeneratedColumn, type Relation, Unique } from 'typeorm';

export enum PostStatus {
  DRAFT = 'draft',
  PUBLISHED = 'published',
}

@Entity('blog_user')
export class BlogUser {
  @PrimaryGeneratedColumn('increment')
  id!: number;

  @OneToMany(() => BlogPost, (entity) => entity.author)
  author_blog_posts!: BlogPost[];

  @OneToMany(() => BlogPost, (entity) => entity.editor)
  editor_blog_posts!: BlogPost[];

  @OneToOne(() => BlogProfile, (entity) => entity.user)
  blog_profile!: Relation<BlogProfile> | null;
}

@Entity('blog_category')
export class BlogCategory {
  @PrimaryGeneratedColumn('increment')
  id!: number;

  @CreateDateColumn({ type: 'timestamptz' })
  created_at!: Date;

  @Column({ type: 'timestamptz' })
  updated_at!: Date;

  @Column({ type: 'varchar', length: 100, unique: true })
  name!: string;

  @Column({ type: 'varchar', length: 50 })
  slug!: string;

  @ManyToOne(() => BlogCategory, (entity) => entity.blog_categories, {
    onDelete: 'SET NULL',
    nullable: true,
  })
  @JoinColumn({ name: 'parent_id' })
  parent!: Relation<BlogCategory> | null;

  @OneToMany(() => BlogCategory, (entity) => entity.parent)
  blog_categories!: BlogCategory[];

  @OneToMany(() => BlogPost, (entity) => entity.category)
  blog_posts!: BlogPost[];
}

@Entity('blog_tag')
export class BlogTag {
  @PrimaryGeneratedColumn('increment')
  id!: number;

  @Column({ type: 'varchar', length: 50 })
  label!: string;

  @ManyToMany(() => BlogPost, (entity) => entity.tags)
  blog_posts!: BlogPost[];
}

@Entity('blog_post')
@Unique('blog_post_author_id_title_key', ['author', 'title'])
@Index('blog_post_title_idx', ['title'])
@Index('post_pub_status_idx', ['published_at', 'status'])
export class BlogPost {
  @PrimaryGeneratedColumn('increment')
  id!: number;

  @CreateDateColumn({ type: 'timestamptz' })
  created_at!: Date;

  @Column({ type: 'timestamptz' })
  updated_at!: Date;

  @Column({ type: 'uuid', unique: true, default: () => 'gen_random_uuid()' })
  public_id!: string;

  @Column({ type: 'varchar', length: 200 })
  title!: string;

  @Column({ type: 'text' })
  body!: string;

  @Column({ type: 'enum', enum: PostStatus, default: PostStatus.DRAFT })
  status!: PostStatus;

  @Column({ type: 'decimal', precision: 4, scale: 2, nullable: true })
  rating!: string | null;

  @Column({ type: 'int', default: 0 })
  view_count!: number;

  @Column({ type: 'boolean', default: false })
  is_featured!: boolean;

  @Column({ type: 'timestamptz', default: () => 'CURRENT_TIMESTAMP' })
  published_at!: Date;

  @Column({ type: 'jsonb', default: () => "'{}'" })
  metadata!: unknown;

  @ManyToOne(() => BlogUser, (entity) => entity.author_blog_posts, {
    onDelete: 'CASCADE',
    nullable: false,
  })
  @JoinColumn({ name: 'author_id' })
  author!: Relation<BlogUser>;

  @ManyToOne(() => BlogUser, (entity) => entity.editor_blog_posts, {
    onDelete: 'SET NULL',
    nullable: true,
  })
  @JoinColumn({ name: 'editor_id' })
  editor!: Relation<BlogUser> | null;

  @ManyToOne(() => BlogCategory, (entity) => entity.blog_posts, {
    onDelete: 'RESTRICT',
    nullable: false,
  })
  @JoinColumn({ name: 'category_id' })
  category!: Relation<BlogCategory>;

  @ManyToMany(() => BlogTag, (entity) => entity.blog_posts)
  @JoinTable({
    name: 'blog_post_tags',
    joinColumn: { name: 'blog_post_id', referencedColumnName: 'id' },
    inverseJoinColumn: { name: 'blog_tag_id', referencedColumnName: 'id' },
  })
  tags!: BlogTag[];
}

@Entity('blog_profile')
export class BlogProfile {
  @PrimaryGeneratedColumn('increment')
  id!: number;

  @Column({ type: 'text', nullable: true })
  bio!: string | null;

  @Column({ type: 'varchar', length: 100, nullable: true })
  avatar!: string | null;

  @OneToOne(() => BlogUser, (entity) => entity.blog_profile, {
    onDelete: 'CASCADE',
    nullable: false,
  })
  @JoinColumn({ name: 'user_id' })
  user!: Relation<BlogUser>;
}
