export enum PostStatus {
  Draft = 'draft',
  Published = 'published',
}

export interface Category {
  createdAt: string;
  updatedAt: string;
  id: number;
  name: string;
  slug: string;
  parent_id: number | null;
  parent?: Category | null;
  children?: Category[];
  post_set?: Post[];
}

export interface Post {
  createdAt: string;
  updatedAt: string;
  id: number;
  publicId: string;
  title: string;
  body: string;
  status: PostStatus;
  rating: string | null;
  viewCount: number;
  isFeatured: boolean;
  publishedAt: string;
  metadata: unknown;
  author_id: number;
  author?: User;
  editor_id: number | null;
  editor?: User | null;
  category_id: number;
  category?: Category;
  tags?: Tag[];
}

export interface Tag {
  id: number;
  label: string;
  posts?: Post[];
}

export interface Profile {
  id: number;
  bio: string | null;
  avatar: string | null;
  user_id: number;
  user?: User;
}

export interface User {
  id: number;
  posts?: Post[];
  editedPosts?: Post[];
  profile?: Profile | null;
}
