import { Column, CreateDateColumn, Entity, Index, JoinColumn, JoinTable, ManyToMany, ManyToOne, OneToMany, OneToOne, PrimaryGeneratedColumn, type Relation, Unique, UpdateDateColumn } from 'typeorm';

export enum PostStatus {
  DRAFT = 'draft',
  PUBLISHED = 'published',
}

@Entity('blog_user')
export class BlogUser {
  @PrimaryGeneratedColumn('uuid')
  id!: string;

  @CreateDateColumn({ name: 'created_at', type: 'timestamptz' })
  createdAt!: Date;

  @UpdateDateColumn({ name: 'updated_at', type: 'timestamptz' })
  updatedAt!: Date;

  @OneToMany(() => BlogPost, (entity) => entity.author)
  authorBlogPosts!: BlogPost[];

  @OneToMany(() => BlogPost, (entity) => entity.editor)
  editorBlogPosts!: BlogPost[];

  @OneToOne(() => BlogProfile, (entity) => entity.user)
  blogProfile!: Relation<BlogProfile> | null;
}

@Entity('blog_category')
export class BlogCategory {
  @PrimaryGeneratedColumn('uuid')
  id!: string;

  @CreateDateColumn({ name: 'created_at', type: 'timestamptz' })
  createdAt!: Date;

  @Column({ name: 'updated_at', type: 'timestamptz' })
  updatedAt!: Date;

  @Column({ type: 'varchar', length: 100, unique: true })
  name!: string;

  @Column({ type: 'varchar', length: 50 })
  slug!: string;

  @ManyToOne(() => BlogCategory, (entity) => entity.blogCategories, {
    onDelete: 'SET NULL',
    nullable: true,
  })
  @JoinColumn({ name: 'parent_id' })
  parent!: Relation<BlogCategory> | null;

  @OneToMany(() => BlogCategory, (entity) => entity.parent)
  blogCategories!: BlogCategory[];

  @OneToMany(() => BlogPost, (entity) => entity.category)
  blogPosts!: BlogPost[];
}

@Entity('blog_tag')
export class BlogTag {
  @PrimaryGeneratedColumn('uuid')
  id!: string;

  @Column({ type: 'varchar', length: 50 })
  label!: string;

  @CreateDateColumn({ name: 'created_at', type: 'timestamptz' })
  createdAt!: Date;

  @UpdateDateColumn({ name: 'updated_at', type: 'timestamptz' })
  updatedAt!: Date;

  @ManyToMany(() => BlogPost, (entity) => entity.tags)
  blogPosts!: BlogPost[];
}

@Entity('blog_post')
@Unique('blog_post_author_id_title_key', ['author', 'title'])
@Index('blog_post_title_idx', ['title'])
@Index('post_pub_status_idx', ['publishedAt', 'status'])
export class BlogPost {
  @PrimaryGeneratedColumn('uuid')
  id!: string;

  @CreateDateColumn({ name: 'created_at', type: 'timestamptz' })
  createdAt!: Date;

  @Column({ name: 'updated_at', type: 'timestamptz' })
  updatedAt!: Date;

  @Column({
    name: 'public_id',
    type: 'uuid',
    unique: true,
    default: () => 'gen_random_uuid()',
  })
  publicId!: string;

  @Column({ type: 'varchar', length: 200 })
  title!: string;

  @Column({ type: 'text' })
  body!: string;

  @Column({ type: 'enum', enum: PostStatus, default: PostStatus.DRAFT })
  status!: PostStatus;

  @Column({ type: 'decimal', precision: 4, scale: 2, nullable: true })
  rating!: string | null;

  @Column({ name: 'view_count', type: 'int', default: 0 })
  viewCount!: number;

  @Column({ name: 'is_featured', type: 'boolean', default: false })
  isFeatured!: boolean;

  @Column({
    name: 'published_at',
    type: 'timestamptz',
    default: () => 'CURRENT_TIMESTAMP',
  })
  publishedAt!: Date;

  @Column({ type: 'jsonb', default: () => "'{}'" })
  metadata!: unknown;

  @ManyToOne(() => BlogUser, (entity) => entity.authorBlogPosts, {
    onDelete: 'CASCADE',
    nullable: false,
  })
  @JoinColumn({ name: 'author_id' })
  author!: Relation<BlogUser>;

  @ManyToOne(() => BlogUser, (entity) => entity.editorBlogPosts, {
    onDelete: 'SET NULL',
    nullable: true,
  })
  @JoinColumn({ name: 'editor_id' })
  editor!: Relation<BlogUser> | null;

  @ManyToOne(() => BlogCategory, (entity) => entity.blogPosts, {
    onDelete: 'RESTRICT',
    nullable: false,
  })
  @JoinColumn({ name: 'category_id' })
  category!: Relation<BlogCategory>;

  @ManyToMany(() => BlogTag, (entity) => entity.blogPosts)
  @JoinTable({
    name: 'blog_post_tags',
    joinColumn: { name: 'blog_post_id', referencedColumnName: 'id' },
    inverseJoinColumn: { name: 'blog_tag_id', referencedColumnName: 'id' },
  })
  tags!: BlogTag[];
}

@Entity('blog_profile')
export class BlogProfile {
  @PrimaryGeneratedColumn('uuid')
  id!: string;

  @Column({ type: 'text', nullable: true })
  bio!: string | null;

  @Column({ type: 'varchar', length: 100, nullable: true })
  avatar!: string | null;

  @CreateDateColumn({ name: 'created_at', type: 'timestamptz' })
  createdAt!: Date;

  @UpdateDateColumn({ name: 'updated_at', type: 'timestamptz' })
  updatedAt!: Date;

  @OneToOne(() => BlogUser, (entity) => entity.blogProfile, {
    onDelete: 'CASCADE',
    nullable: false,
  })
  @JoinColumn({ name: 'user_id' })
  user!: Relation<BlogUser>;
}
