export enum PostStatus {
  Draft = 'draft',
  Published = 'published',
}

export interface User {
  id: number;
  posts?: Post[];
  editedPosts?: Post[];
  profile?: Profile | null;
}

export interface Category {
  id: number;
  created_at: string;
  updated_at: string;
  name: string;
  slug: string;
  parent_id: number | null;
  parent?: Category | null;
  children?: Category[];
  posts?: Post[];
}

export interface Tag {
  id: number;
  label: string;
  posts?: Post[];
}

export interface Post {
  id: number;
  created_at: string;
  updated_at: string;
  public_id: string;
  title: string;
  body: string;
  status: PostStatus;
  rating: string | null;
  view_count: number;
  is_featured: boolean;
  published_at: string;
  metadata: unknown;
  author_id: number;
  author?: User;
  editor_id: number | null;
  editor?: User | null;
  category_id: number;
  category?: Category;
  tags?: Tag[];
}

export interface Profile {
  id: number;
  bio: string | null;
  avatar: string | null;
  user_id: number;
  user?: User;
}
