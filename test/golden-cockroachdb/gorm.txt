=== models/account.go
package models

import (
	"time"

	"github.com/google/uuid"
	"github.com/shopspring/decimal"
	"gorm.io/datatypes"
)

// Account maps to the "accounts" table.
type Account struct {
	ID         int64                       `gorm:"primaryKey;autoIncrement"`
	Email      string                      `gorm:"size:255;not null;unique;uniqueIndex:uni_accounts_email_status,priority:1;index"`
	Age        *int32                      `gorm:"index"`
	Visits     int64                       `gorm:"not null;default:0;index"`
	Balance    decimal.Decimal             `gorm:"type:decimal(12,2);not null"`
	Bio        *string                     `gorm:"type:text"`
	Active     bool                        `gorm:"not null;default:true"`
	CreatedAt  time.Time                   `gorm:"not null;autoCreateTime;index"`
	UpdatedAt  time.Time                   `gorm:"not null;autoUpdateTime"`
	Prefs      datatypes.JSON              `gorm:"index"`
	Labels     datatypes.JSONSlice[string] `gorm:"not null;index"`
	Grid       datatypes.JSONSlice[[]int32]
	Attributes datatypes.JSON `gorm:"not null;default:'{}'"`
	StockRange *string        `gorm:"index"`
	LastIP     *string        `gorm:"size:45"`
	ShelfLife  *time.Duration
	Avatar     []byte
	Status     Status    `gorm:"size:10;not null;default:draft;uniqueIndex:uni_accounts_email_status,priority:2"`
	TenantID   uuid.UUID `gorm:"type:uuid;not null"`

	Tenant Tenant  `gorm:"foreignKey:TenantID;constraint:OnDelete:CASCADE"`
	Events []Event `gorm:"foreignKey:AccountID;constraint:OnDelete:SET NULL"`
}

=== models/event.go
package models

import "gorm.io/datatypes"

// Event maps to the "events" table.
type Event struct {
	ID        int64          `gorm:"primaryKey;autoIncrement"`
	Kind      int32          `gorm:"not null;default:0"`
	Payload   datatypes.JSON `gorm:"not null;default:'{}'"`
	AccountID *int64

	Account *Account `gorm:"foreignKey:AccountID;constraint:OnDelete:SET NULL"`
}

=== models/tenant.go
package models

import "github.com/google/uuid"

// Tenant maps to the "tenants" table.
type Tenant struct {
	ID     uuid.UUID `gorm:"type:uuid;primaryKey;default:gen_random_uuid()"`
	Name   string    `gorm:"size:80;not null;unique"`
	Status Status    `gorm:"size:10;not null;default:draft"`

	Accounts []Account `gorm:"foreignKey:TenantID;constraint:OnDelete:CASCADE"`
}
