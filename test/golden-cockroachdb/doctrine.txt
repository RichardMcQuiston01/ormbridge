namespace App\Entity;

use App\Enum\Status;
use Doctrine\Common\Collections\ArrayCollection;
use Doctrine\Common\Collections\Collection;
use Doctrine\ORM\Mapping as ORM;

#[ORM\Entity]
#[ORM\Table(name: 'tenants')]
class Tenant
{
    #[ORM\Id]
    #[ORM\Column(name: 'id', type: 'guid')]
    private string $id;

    #[ORM\Column(name: 'name', type: 'string', length: 80, unique: true)]
    private string $name;

    #[ORM\Column(
        name: 'status',
        type: 'string',
        length: 10,
        enumType: Status::class,
        options: ['default' => 'draft'],
    )]
    private Status $status = Status::DRAFT;

    /** @var Collection<int, Account> */
    #[ORM\OneToMany(targetEntity: Account::class, mappedBy: 'tenant')]
    private Collection $account_set;

    public function __construct()
    {
        $this->id = self::generateUuid();
        $this->account_set = new ArrayCollection();
    }

    public function getId(): string
    {
        return $this->id;
    }

    public function setId(string $id): static
    {
        $this->id = $id;

        return $this;
    }

    public function getName(): string
    {
        return $this->name;
    }

    public function setName(string $name): static
    {
        $this->name = $name;

        return $this;
    }

    public function getStatus(): Status
    {
        return $this->status;
    }

    public function setStatus(Status $status): static
    {
        $this->status = $status;

        return $this;
    }

    /**
     * @return Collection<int, Account>
     */
    public function getAccountSet(): Collection
    {
        return $this->account_set;
    }

    public function addAccount(Account $account): static
    {
        if (!$this->account_set->contains($account)) {
            $this->account_set->add($account);
            $account->setTenant($this);
        }

        return $this;
    }

    public function removeAccount(Account $account): static
    {
        if ($this->account_set->removeElement($account)) {
            // The owning side is required, so it is not unset here.
        }

        return $this;
    }

    private static function generateUuid(): string
    {
        $bytes = random_bytes(16);
        $bytes[6] = chr((ord($bytes[6]) & 0x0f) | 0x40);
        $bytes[8] = chr((ord($bytes[8]) & 0x3f) | 0x80);

        return vsprintf('%s%s-%s-%s-%s-%s%s%s', str_split(bin2hex($bytes), 4));
    }
}
